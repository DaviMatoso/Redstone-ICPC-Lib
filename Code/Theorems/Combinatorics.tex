\subsection{Combinatorics}

\begin{small}
\begin{itemize}
    \item \textbf{Sums:}
    \begin{center}
    \[
    \begin{array}{@{}l@{\qquad}l@{}}
        \sum_{k=0}^n k = \frac{n(n+1)}{2}
            & {n \choose k} = \frac{n!}{(n-k)!k!} \\[2pt]

        \sum_{k=a}^b k = \frac{(a+b)(b-a+1)}{2}
            & {n \choose k} = {n-1 \choose k} + {n-1 \choose k-1} \\[2pt]

        \sum_{k=0}^n k^2 = \frac{n(n+1)(2n+1)}{6}
            & {n+1 \choose k} = \frac{n+1}{n-k+1}{n \choose k} \\[2pt]

        \sum_{k=0}^n k^3 = \frac{n^2(n+1)^2}{4}
            & {n \choose k+1} = \frac{n-k}{k+1}{n \choose k} \\[2pt]

        \sum_{k=0}^n k^4 =
            \frac{6n^5+15n^4+10n^3-n}{30}
            & {n \choose k} = \frac{n}{n-k}{n-1 \choose k} \\[2pt]

        \sum_{k=0}^n k^5 =
            \frac{2n^6+6n^5+5n^4-n^2}{12}
            & {n \choose k} = \frac{n-k+1}{k}{n \choose k-1} \\[2pt]

        \sum_{k=0}^n x^k = \frac{x^{n+1}-1}{x-1}
            & 12! \approx 2^{28.8} \\[2pt]

        \sum_{k=0}^n kx^k =
            \frac{x-(n+1)x^{n+1}+nx^{n+2}}{(x-1)^2}
            & 20! \approx 2^{61.1} \\[2pt]

        1+x+x^2+\dots = \frac{1}{1-x}
            &
    \end{array}
    \]
    \end{center}
    \begin{itemize}
        \item \textbf{Hockey-stick identity:} $\sum_{i=r}^{n}\binom{i}{r} = \binom{n+1}{r+1}$
        \item \textbf{Number of ways to color $n$-objects with $r$-colors} if all colors must be used at least once:
        \[ \sum_{k=0}^r {r \choose k}(-1)^{r-k}k^n \quad \text{or} \quad \sum_{k=0}^r {r \choose r-k}(-1)^{k}(r-k)^n \]
    \end{itemize}

    \item \textbf{Binomial coefficients:}
    \begin{itemize}
        \item Number of ways to pick a multiset of size $k$ from $n$ elements: ${n+k-1 \choose k}$
        \item Number of $n$-tuples of non-negative integers with sum $s$: ${s+n-1 \choose n-1}$, at most $s$: ${s + n \choose n}$
        \item Number of $n$-tuples of positive integers with sum $s$: ${s-1 \choose n-1}$
        \item Number of lattice paths from $(0,0)$ to $(a,b)$, restricted to east and north steps: ${a+b \choose a}$
    \end{itemize}

    \item \textbf{Multinomial theorem:}
    $(a_1+\dots+a_k)^n = \sum {n \choose n_1,\dots,n_k} a_1^{n_1} \dots a_k^{n_k}$, where $n_i \ge 0$ and $\sum n_i=n$.
    \[ {n \choose n_1,\dots,n_k} = M(n_1,\dots,n_k) = \frac{n!}{n_1! \dots n_k!} \]
    \[ M(a,\dots,b,c,\dots) = M(a+\dots+b,c,\dots) M(a,\dots,b) \]

    \item \textbf{Catalan numbers:}
    \begin{itemize}
        \item $C_n = \frac{1}{n+1} \binom{2n}{n} = \frac{(2n)!}{(n+1)!n!}$ with $n \geq 0$, $C_0 = 1$ and $C_{n+1} = \frac{2(2n + 1)}{n + 2} C_{n}$. Also, $C_n=\sum_{i=0}^{n-1} C_i C_{n-1-i}$    
        \item Sequence: 1, 1, 2, 5, 14, 42, 132, 429, 1430, 4862, 16796, 58786, 208012, 742900, 2674440, 9694845, 35357670
        \item $C_n$ is the number of: properly nested sequences of $n$ pairs of parentheses; rooted ordered binary trees with $n+1$ leaves; triangulations of a convex $(n+2)$-gon.
    \end{itemize}

    \item \textbf{Derangements:}
    Number of permutations of $n=0,1,2,\dots$ elements without fixed points is $1, 0, 1, 2, 9, 44, 265, 1854, 14833, \dots$ \\
    Recurrence: $D_n = (n-1)(D_{n-1} + D_{n-2}) = n D_{n-1} + (-1)^n$. \\
    Corollary: number of permutations with exactly $k$ fixed points is ${n \choose k} D_{n-k}$.

    \item \textbf{Stirling numbers of $1^{st}$ kind:}
    $s_{n,k}$ is $(-1)^{n-k}$ times the number of permutations of $n$ elements with exactly $k$ permutation cycles. \\
    $|s_{n,k}| = |s_{n-1,k-1}| + (n-1) |s_{n-1,k}|$. \quad
    $\sum_{k=0}^n s_{n,k}\,x^k = x^{\underline n}$

    \item \textbf{Stirling numbers of $2^{nd}$ kind:}
    $S_{n,k}$ is the number of ways to partition a set of $n$ elements into exactly $k$ non-empty subsets. \\
    $S_{n,k} = S_{n-1,k-1} + k S_{n-1,k}$. \quad $S_{n,1} = S_{n,n} = 1$. \quad
    $x^n = \sum_{k=0}^n S_{n,k}\,x^{\underline k}$

    \item \textbf{Bell numbers:}
    $B_n$ is the number of partitions of $n$ elements. \\
    $B_0, \ldots = 1,1,2,5,15,52,203,\ldots$ \\
    $B_{n+1} = \sum_{k=0}^n {n \choose k} B_k = \sum_{k=1}^n S_{n,k}$. \\
    Bell triangle: $B_r=a_{r,1}=a_{r-1,r-1}$, $a_{r,c}=a_{r-1,c-1}+a_{r,c-1}$.

    \item \textbf{Bernoulli numbers:}
    $\sum_{k=0}^{m-1} k^n = \frac{1}{n+1} \sum_{k=0}^n {n+1 \choose k} B_k m^{n+1-k}$. \\
    $\sum_{j=0}^m {m+1 \choose j} B_j = 0$. \quad $B_0=1$, $B_1=-\frac{1}{2}$. \quad $B_n=0$, for all odd $n \ne 1$.

    \item \textbf{Eulerian numbers:}
    $E(n,k)$ is the number of permutations with exactly $k$ descents ($i: \pi_i < \pi_{i+1}$) / ascents ($\pi_i > \pi_{i+1}$) / excedances ($\pi_i > i$) / $k+1$ weak excedances ($\pi_i \ge i$). \\
    Formula: $E(n,k)=(k+1)E(n-1,k)+(n-k)E(n-1,k-1)$. \quad
    $x^n = \sum_{k=0}^{n-1} E(n,k) {x+k \choose n}$.

    \item \textbf{Burnside's lemma:}
    The number of orbits under group $G$'s action on set $X$:
    \[ |X/G| = \frac{1}{|G|} \sum_{g \in G} |X_g| \]
    where $X_g=\{ x \in X: g(x)=x \}$ (``Average number of fixed points.'') \\
    Let $w(x)$ be weight of $x$'s orbit. Sum of all orbits' weights:
    \[ \sum_{o \in X/G} w(o) = \frac{1}{|G|} \sum_{g \in G} \sum_{x \in X_g} w(x) \]
\end{itemize}
\end{small}
